   220	def _verify_migration(conn: sqlite3.Connection) -> None:
   221	    unversioned_sessions = conn.execute(
   222	        """SELECT COUNT(*) FROM sessions
   223	           WHERE question_bank_version IS NULL
   224	              OR trim(question_bank_version)=''"""
   225	    ).fetchone()[0]
   226	    if unversioned_sessions:
   227	        raise ValueError("Every session must have a question bank version")
   228	
   229	    orphaned_responses = conn.execute(
   230	        """SELECT COUNT(*) FROM responses r
   231	           LEFT JOIN question_bank_items qbi
   232	             ON qbi.question_bank_version=?
   233	            AND qbi.source_item_id=r.item_id
   234	           LEFT JOIN session_responses sr
   235	             ON sr.session_id=r.session_id
   236	            AND sr.item_record_id=qbi.item_record_id
   237	           WHERE sr.item_record_id IS NULL""",
   238	        (LEGACY_BANK_VERSION,),
   239	    ).fetchone()[0]
   240	    if orphaned_responses:
   241	        raise ValueError(
   242	            f"Migration left {orphaned_responses} legacy responses unlinked"
   243	        )
   244	
   245	    foreign_key_errors = conn.execute(
   246	        "PRAGMA foreign_key_check"
   247	    ).fetchall()
   248	    if foreign_key_errors:
   249	        raise ValueError("Database contains foreign key errors after migration")

   252	    initialize()

-------------------- innergeodessa-mvp/backend/app/scoring.py --------------------
     1	from __future__ import annotations
     2	from dataclasses import dataclass
     3	from typing import Iterable, Mapping
     4	
     5	DIMENSIONS = {
     6	    "EI": ("E", "I"),
     7	    "SN": ("S", "N"),
     8	    "TF": ("T", "F"),
     9	    "JP": ("J", "P"),
    10	}
    11	
    12	TIE_RULE = (
    13	    "A zero score resolves to the first pole and must be "
    14	    "reported as low differentiation."
    15	)
    16	
    17	@dataclass(frozen=True)
    18	class Item:
    19	    item_id: str
    20	    dimension: str
    21	    key: str
    22	
    23	def score_assessment(
    24	    items: Iterable[Item],
    25	    responses: Mapping[str, int],
    26	    *,
    27	    require_balanced_poles: bool = True,
    28	) -> dict:
    29	    item_list = list(items)
    30	    _validate_items(item_list, require_balanced_poles)
    31	    _validate_responses(item_list, responses)
    32	
    33	    signed = {dimension: 0 for dimension in DIMENSIONS}
    34	    answered = {dimension: 0 for dimension in DIMENSIONS}
    35	
    36	    for item in item_list:
    37	        raw = responses[item.item_id]
    38	
    39	        first, second = DIMENSIONS[item.dimension]
    40	        contribution = raw - 3 if item.key == first else 3 - raw
    41	        signed[item.dimension] += contribution
    42	        answered[item.dimension] += 1
    43	
    44	    letters = []
    45	    confidence = {}
    46	    for dimension, (first, second) in DIMENSIONS.items():
    47	        value = signed[dimension]
    48	        letters.append(first if value >= 0 else second)
    49	        max_abs = answered[dimension] * 2
    50	        confidence[dimension] = round(abs(value) / max_abs, 4) if max_abs else 0.0
    51	
    52	    return {
    53	        "type": "".join(letters),
    54	        "scores": signed,
    55	        "confidence": confidence,
    56	        "answered": answered,
    57	        "tie_rule": TIE_RULE,
    58	    }
    59	
    60	
    61	def _validate_items(
    62	    items: list[Item],
    63	    require_balanced_poles: bool,
    64	) -> None:
    65	    item_ids: set[str] = set()
    66	
    67	    for item in items:
    68	        if not item.item_id:
    69	            raise ValueError("item_id must not be empty")
    70	        if item.item_id in item_ids:
    71	            raise ValueError(f"Duplicate item_id: {item.item_id}")
    72	        item_ids.add(item.item_id)
    73	
    74	    if len(items) != 72:
    75	        raise ValueError(
    76	            f"Personality assessment must contain exactly 72 items; found {len(items)}"
    77	        )
    78	
    79	    pole_counts = {
    80	        dimension: {first: 0, second: 0}
    81	        for dimension, (first, second) in DIMENSIONS.items()
    82	    }
    83	
    84	    for item in items:
    85	        if item.dimension not in DIMENSIONS:
    86	            raise ValueError(f"Unsupported dimension: {item.dimension}")
    87	
    88	        first, second = DIMENSIONS[item.dimension]
    89	        if item.key not in (first, second):
    90	            raise ValueError(
    91	                f"Invalid key {item.key} for dimension {item.dimension}; "
    92	                f"expected {first} or {second}"
    93	            )
    94	
    95	        pole_counts[item.dimension][item.key] += 1
    96	
    97	    for dimension, (first, second) in DIMENSIONS.items():
    98	        first_count = pole_counts[dimension][first]
    99	        second_count = pole_counts[dimension][second]
   100	        dimension_total = first_count + second_count
   101	        if dimension_total != 18:
   102	            raise ValueError(
   103	                f"{dimension} must contain exactly 18 items; "
   104	                f"found {dimension_total}"
   105	            )
   106	        if (
   107	            require_balanced_poles
   108	            and (first_count != 9 or second_count != 9)
   109	        ):
   110	            raise ValueError(
   111	                f"{dimension} must contain 9 {first} and 9 {second} items; "
   112	                f"found {first_count} and {second_count}"
   113	            )
   114	
   115	
   116	def _validate_responses(
   117	    items: list[Item],
   118	    responses: Mapping[str, int],
   119	) -> None:
   120	    item_ids = {item.item_id for item in items}
   121	    response_ids = set(responses)
   122	
   123	    unknown = sorted(response_ids - item_ids)
   124	    if unknown:
   125	        raise ValueError(f"Unknown response item: {unknown[0]}")
   126	
   127	    missing = sorted(item_ids - response_ids)
   128	    if missing:
   129	        raise ValueError(f"Missing response for {missing[0]}")
   130	
   131	    for item in items:
   132	        raw = responses[item.item_id]
   133	        if type(raw) is not int or raw not in (1, 2, 3, 4, 5):
   134	            raise ValueError(
   135	                f"Response for {item.item_id} must be an integer from 1-5"
   136	            )
