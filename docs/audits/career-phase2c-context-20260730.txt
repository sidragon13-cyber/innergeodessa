     1	import {
     2	  RIASEC_DIMENSIONS,
     3	  type RiasecDimension,
     4	  type RiasecPercentages,
     5	  type RiasecScores,
     6	} from "./types";
     7	
     8	export interface RiasecResultContract {
     9	  sessionId: string;
    10	  module: "riasec";
    11	  status: "completed";
    12	  code: string;
    13	  scores: RiasecScores;
    14	  percentages: RiasecPercentages;
    15	  ranking: RiasecDimension[];
    16	  answered: RiasecScores;
    17	  questionBankVersion: string;
    18	  completedAt: string;
    19	  calculatedAt: string;
    20	}
    21	
    22	export function isRiasecResultContract(
    23	  value: unknown,
    24	): value is RiasecResultContract {
    25	  if (
    26	    !isRecord(value) ||
    27	    !isNonEmptyString(value.sessionId) ||
    28	    value.module !== "riasec" ||
    29	    value.status !== "completed" ||
    30	    !isNonEmptyString(value.code) ||
    31	    !isDimensionRecord(value.scores, isFiniteNumber) ||
    32	    !isDimensionRecord(
    33	      value.percentages,
    34	      isValidPercentage,
    35	    ) ||
    36	    !isRanking(value.ranking) ||
    37	    !isDimensionRecord(
    38	      value.answered,
    39	      isNonNegativeInteger,
    40	    ) ||
    41	    !isNonEmptyString(value.questionBankVersion) ||
    42	    !isNonEmptyString(value.completedAt) ||
    43	    !isNonEmptyString(value.calculatedAt)
    44	  ) {
    45	    return false;
    46	  }
    47	
    48	  return value.code === value.ranking.slice(0, 3).join("");
    49	}
    50	
    51	function isRanking(value: unknown): value is RiasecDimension[] {
    52	  return (
    53	    Array.isArray(value) &&
    54	    value.length === RIASEC_DIMENSIONS.length &&
    55	    value.every(isRiasecDimension) &&
    56	    new Set(value).size === RIASEC_DIMENSIONS.length
    57	  );
    58	}
    59	
    60	function isDimensionRecord(
    61	  value: unknown,
    62	  validateValue: (entry: unknown) => boolean,
    63	): boolean {
    64	  return (
    65	    isRecord(value) &&
    66	    RIASEC_DIMENSIONS.every((dimension) =>
    67	      validateValue(value[dimension]),
    68	    )
    69	  );
    70	}
    71	
    72	function isRiasecDimension(
    73	  value: unknown,
    74	): value is RiasecDimension {
    75	  return (
    76	    typeof value === "string" &&
    77	    RIASEC_DIMENSIONS.some((dimension) => dimension === value)
    78	  );
    79	}
    80	
    81	function isFiniteNumber(value: unknown): boolean {
    82	  return typeof value === "number" && Number.isFinite(value);
    83	}
    84	
    85	function isValidPercentage(value: unknown): boolean {
    86	  return (
    87	    isFiniteNumber(value) &&
    88	    typeof value === "number" &&
    89	    value >= 0 &&
    90	    value <= 100
    91	  );
    92	}
    93	
    94	function isNonNegativeInteger(value: unknown): boolean {
    95	  return (
    96	    typeof value === "number" &&
    97	    Number.isInteger(value) &&
    98	    value >= 0
    99	  );
   100	}
   101	
   102	function isNonEmptyString(value: unknown): value is string {
   103	  return typeof value === "string" && value.trim().length > 0;

    19	@router.post("/api/sessions")

    21	    if not payload.consent:
    22	        raise HTTPException(400, "Consent is required.")
    23	
    24	    try:
    25	        module_config = get_assessment_module_config(payload.module)
    26	    except ValueError as error:
    27	        raise HTTPException(400, str(error)) from error

    29	    session_id = str(uuid.uuid4())

    31	
    32	    with connect() as conn:
    33	        conn.execute(
    34	            """INSERT INTO sessions(
    35	                 session_id, consent, language, started_at,
    36	                 question_bank_version, module
    37	               )
    38	               VALUES (?,1,?,?,?,?)""",
    39	            (
    40	                session_id,
    41	                payload.language,
    42	                now,
    43	                module_config.question_bank_version,
    44	                module_config.module,
    45	            ),
    46	        )
    47	        if module_config.module == "riasec":
    48	            snapshot_table = "riasec_session_question_items"
    49	            item_table = "riasec_question_items"
    50	        else:
    51	            snapshot_table = "session_question_items"
    52	            item_table = "question_bank_items"
    53	
    54	        conn.execute(
    55	            f"""INSERT INTO {snapshot_table}(
    56	                  session_id, item_record_id, display_order
    57	                )
    58	                SELECT ?, item_record_id, master_order
    59	                FROM {item_table}
    60	                WHERE question_bank_version=?""",
    61	            (session_id, module_config.question_bank_version),
    62	        )
    63	
    64	        snapshot_count = conn.execute(
    65	            f"""SELECT COUNT(*)
    66	                FROM {snapshot_table}
    67	                WHERE session_id=?""",
    68	            (session_id,),
    69	        ).fetchone()[0]
    70	
    71	        if snapshot_count != module_config.expected_item_count:
    72	            raise HTTPException(
    73	                503,
    74	                detail={
    75	                    "message": (
    76	                        "Assessment question bank is unavailable."
    77	                    ),
    78	                    "module": module_config.module,
    79	                    "question_bank_version": (
    80	                        module_config.question_bank_version
    81	                    ),
    82	                    "expected": module_config.expected_item_count,
    83	                    "actual": snapshot_count,
    84	                },
    85	            )
    86	
    87	    return {
    88	        "session_id": session_id,
    89	        "started_at": now,
    90	        "module": module_config.module,
    91	        "question_bank_version": (
    92	            module_config.question_bank_version
    93	        ),
    94	    }

    97	@router.get("/api/sessions/{session_id}/items")
    98	def get_session_items(session_id: str):
    99	    with connect() as conn:
   100	        session = conn.execute(
   101	            """SELECT question_bank_version, module
   102	               FROM sessions WHERE session_id=?""",
   103	            (session_id,),
   104	        ).fetchone()
   105	        if not session:
   106	            raise HTTPException(404, "Session not found.")
   107	
   108	        if session["module"] == "riasec":
   109	            rows = conn.execute(
   110	                """SELECT item.source_item_id AS item_id,
   111	                          item.dimension,
   112	                          item.wording,
   113	                          snapshot.display_order AS master_order,
   114	                          item.question_bank_version
   115	                   FROM riasec_session_question_items snapshot
   116	                   JOIN riasec_question_items item
   117	                     ON item.item_record_id=snapshot.item_record_id
   118	                   WHERE snapshot.session_id=?
   119	                   ORDER BY snapshot.display_order""",
   120	                (session_id,),
   121	            ).fetchall()
   122	        elif session["module"] == "personality":
   123	            rows = conn.execute(
   124	                """SELECT qbi.source_item_id AS item_id,
   125	                          qbi.dimension,
   126	                          qbi.subdimension,
   127	                          qbi.keyed_pole,
   128	                          qbi.form,
   129	                          qbi.wording,
   130	                          sqi.display_order AS master_order,
   131	                          qbi.question_bank_version
   132	                   FROM session_question_items sqi
   133	                   JOIN question_bank_items qbi
   134	                     ON qbi.item_record_id=sqi.item_record_id
   135	                   WHERE sqi.session_id=?
   136	                   ORDER BY sqi.display_order""",
   137	                (session_id,),
   138	            ).fetchall()
   139	        else:
   140	            raise HTTPException(400, "Session module is not supported.")
   141	    return [dict(row) for row in rows]

   240	@router.put("/api/sessions/{session_id}/answers")
   241	def save_answer(session_id: str, payload: AnswerRequest):
   242	    now = datetime.now(timezone.utc).isoformat()
   243	    with connect() as conn:
   244	        session = conn.execute(
   245	            "SELECT status, module FROM sessions WHERE session_id=?",
   246	            (session_id,),
   247	        ).fetchone()
   248	        if not session:
   249	            raise HTTPException(404, "Session not found.")
   250	        if session["status"] != "active":
   251	            raise HTTPException(409, "Session is not active.")
   252	        if session["module"] == "riasec":
   253	            item = conn.execute(
   254	                """SELECT snapshot.item_record_id
   255	                   FROM riasec_session_question_items snapshot
   256	                   JOIN riasec_question_items item
   257	                     ON item.item_record_id=snapshot.item_record_id
   258	                   WHERE snapshot.session_id=?
   259	                     AND item.source_item_id=?""",
   260	                (session_id, payload.item_id),
   261	            ).fetchone()
   262	            response_table = "riasec_session_responses"
   263	        elif session["module"] == "personality":
   264	            item = conn.execute(
   265	                """SELECT sqi.item_record_id
   266	                   FROM session_question_items sqi
   267	                   JOIN question_bank_items qbi
   268	                     ON qbi.item_record_id=sqi.item_record_id
   269	                   WHERE sqi.session_id=? AND qbi.source_item_id=?""",
   270	                (session_id, payload.item_id),
   271	            ).fetchone()
   272	            response_table = "session_responses"
   273	        else:
   274	            raise HTTPException(400, "Session module is not supported.")
   275	        if not item:
   276	            raise HTTPException(
   277	                404,
   278	                "Item is not part of this session's question snapshot.",
   279	            )
   280	        conn.execute(
   281	            f"""INSERT INTO {response_table}(
   282	                  session_id, item_record_id, raw_value,
   283	                  response_time_ms, answered_at
   284	                )
   285	                VALUES (?,?,?,?,?)
   286	                ON CONFLICT(session_id,item_record_id) DO UPDATE SET
   287	                raw_value=excluded.raw_value,
   288	                response_time_ms=excluded.response_time_ms,
   289	                answered_at=excluded.answered_at""",
   290	            (
   291	                session_id,
   292	                item["item_record_id"],
   293	                payload.value,
   294	                payload.response_time_ms,
   295	                now,
   296	            ),
   297	        )
   298	    return {"saved": True}
