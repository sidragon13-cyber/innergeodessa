     1	export function createReportSectionAnchor(
     2	  order: number,
     3	  identifier: string,
     4	): string {
     5	  const safeOrder = Number.isFinite(order)
     6	    ? Math.max(0, Math.trunc(order))
     7	    : 0;
     8	  const slug = identifier
     9	    .normalize("NFKD")
    10	    .replace(/[\u0300-\u036f]/g, "")
    11	    .toLowerCase()
    12	    .replace(/[^a-z0-9]+/g, "-")
    13	    .replace(/^-+|-+$/g, "");
    14	
    15	  return `report-section-${String(safeOrder).padStart(2, "0")}-${slug || "section"}`;
    16	}

    16	def connect(db_path: Path = DB_PATH) -> sqlite3.Connection:
    17	    conn = sqlite3.connect(db_path)
    18	    conn.row_factory = sqlite3.Row
    19	    conn.execute("PRAGMA foreign_keys = ON")
    20	    return conn

    24	def initialize(
    25	    db_path: Path = DB_PATH,
    26	    items_path: Path = ITEMS_PATH,
    27	) -> None:
    28	    items = _load_current_items(items_path)
    29	
    30	    with connect(db_path) as conn:
    31	        conn.execute("BEGIN IMMEDIATE")
    32	        try:
    33	            _apply_schema(conn)
    34	            _ensure_session_version_column(conn)
    35	            _register_question_banks(conn)
    36	            _capture_legacy_items(conn)
    37	            _backfill_legacy_sessions(conn)
    38	            _create_missing_session_snapshots(conn)
    39	            _migrate_legacy_responses(conn)
    40	            _upsert_current_items(conn, items)
    41	            _verify_migration(conn)
    42	        except Exception:
    43	            conn.rollback()
    44	            raise
    45	        else:
    46	            conn.commit()

    50	    statements = SCHEMA_PATH.read_text(encoding="utf-8").split(";")

    53	            conn.execute(statement)

    56	def _ensure_session_version_column(conn: sqlite3.Connection) -> None:
    57	    columns = {
    58	        row["name"]
    59	        for row in conn.execute("PRAGMA table_info(sessions)").fetchall()
    60	    }
    61	    if "question_bank_version" not in columns:
    62	        conn.execute(
    63	            "ALTER TABLE sessions ADD COLUMN question_bank_version TEXT"
    64	        )

    67	def _load_current_items(items_path: Path) -> list[dict]:
    68	    parsed = json.loads(items_path.read_text(encoding="utf-8"))
    69	    if not isinstance(parsed, list) or len(parsed) != 72:
    70	        raise ValueError("Current personality items must contain exactly 72 rows")
    71	
    72	    required = {
    73	        "item_id",
    74	        "dimension",
    75	        "subdimension",
    76	        "key",
    77	        "form",
    78	        "wording",
    79	        "language",
    80	        "version",
    81	        "question_bank_version",
    82	        "status",
    83	        "master_order",
    84	        "reverse_scored",
    85	    }
    86	    for index, item in enumerate(parsed):
    87	        if not isinstance(item, dict) or not required.issubset(item):
    88	            raise ValueError(f"Current personality item {index} is invalid")
    89	        if item["question_bank_version"] != CURRENT_BANK_VERSION:
    90	            raise ValueError(
    91	                f"Current personality item {index} has an unexpected bank version"
    92	            )
    93	    return parsed

   220	def _verify_migration(conn: sqlite3.Connection) -> None:
   221	    unversioned_sessions = conn.execute(
   222	        """SELECT COUNT(*) FROM sessions
   223	           WHERE question_bank_version IS NULL
   224	              OR trim(question_bank_version)=''"""
   225	    ).fetchone()[0]
   226	    if unversioned_sessions:
   227	        raise ValueError("Every session must have a question bank version")
   228	
   229	    orphaned_responses = conn.execute(
   230	        """SELECT COUNT(*) FROM responses r
   231	           LEFT JOIN question_bank_items qbi
   232	             ON qbi.question_bank_version=?
   233	            AND qbi.source_item_id=r.item_id
   234	           LEFT JOIN session_responses sr
   235	             ON sr.session_id=r.session_id
   236	            AND sr.item_record_id=qbi.item_record_id
   237	           WHERE sr.item_record_id IS NULL""",
   238	        (LEGACY_BANK_VERSION,),
   239	    ).fetchone()[0]
   240	    if orphaned_responses:
   241	        raise ValueError(
   242	            f"Migration left {orphaned_responses} legacy responses unlinked"
   243	        )
   244	
   245	    foreign_key_errors = conn.execute(
   246	        "PRAGMA foreign_key_check"
   247	    ).fetchall()
   248	    if foreign_key_errors:
   249	        raise ValueError("Database contains foreign key errors after migration")

   252	    initialize()
